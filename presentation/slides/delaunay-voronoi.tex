\begin{frame}{The Delaunay-Voronoi Relationship}
	\begin{itemize}
		\item Voronoi verices are equidistant to at least 3 points in the set
		\item Circles centered at Voronoi vertices obey the empty circle property
		\item Connecting these points yields a triangle in the Delaunay triangulation
		\item The Voronoi diagram and the Delaunay Triangulation are \textit{dual graphs} of one another
	\end{itemize}
\end{frame}
