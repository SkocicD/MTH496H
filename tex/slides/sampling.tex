\begin{frame}{Sampling}
	\begin{itemize}
		\item We say that a curve is r-sampled if the distance between any pair of adjacent samples is less than or equal to $r\cdot LFS(p)$
		\item This means that a more detailed section of a curve has samples very close together, while less detailed sections can be far apart
		\item An r-sampled curve with $r\le 2.53$ is guaranteed to be successfully reconstructed by the crust
	\end{itemize}
\end{frame}
