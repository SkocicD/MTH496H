\begin{frame}{The Crust}
	\begin{itemize}
		\item The crust builds on the previously mentioned geometric tools:
		      \begin{enumerate}
			      \item Start with a set of points $S$ and compute the Voronoi Vertices $V$.
			      \item Let $S'=S\cup V$ and let $E$ be the set of edges of the Delaunay triangulation of $S'$
			      \item All edges in $E$ that connect two points of $S$ make up the crust
		      \end{enumerate}
	\end{itemize}
\end{frame}
