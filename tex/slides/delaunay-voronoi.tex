\begin{frame}{The Delaunay-Voronoi Relationship}
	\begin{itemize}
		\item Voronoi vertices are equidistant to at least 3 sites
		\item A circle centered at a Voronoi vertex connecting these sites contains no other sites
		\item Drawing lines between these sites yields a triangle that satisfies the empty circle property
		\item The Voronoi diagram and the Delaunay Triangulation are \textbf{dual graphs} of one another
	\end{itemize}
